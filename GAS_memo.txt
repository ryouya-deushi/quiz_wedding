function doPost(e) {
  try {
    const sheet = SpreadsheetApp.openById('XXXX').getSheetByName('result');
    const data = JSON.parse(e.postData.contents);
    
    // A列の最終行を取得（A列が空でない最後の行）
    const dataRange = sheet.getRange("A:A").getValues();
    let nextRow = dataRange.findIndex(row => !row[0]) + 1; // 最初の空白行を見つける
    
    // 空行がない場合、次の行を追加
    if (nextRow === 0) {
      nextRow = sheet.getLastRow() + 1;
    }
    
    // データの配列を作成
    const row = [
      new Date(),
      data.YourName,
      data.Q1,
      data.Q2,
      data.Q3,
      data.Q4,
      data.Q5,
      data.Q6,
      data.Q7,
      data.Q8,
      data.Q9,
      data.Q10
    ];
    
    // 見つけた空行にデータを書き込む
    sheet.getRange(nextRow, 1, 1, row.length).setValues([row]);

    return ContentService.createTextOutput(JSON.stringify({ result: true }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ result: false, error: error.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}




function doGet(e) {
  var sheet = SpreadsheetApp.openById('XXXX').getSheetByName("passwd");
  var cellValue = sheet.getRange("A1").getValue();
  
  var output = ContentService.createTextOutput(JSON.stringify({ cellValue })).setMimeType(ContentService.MimeType.JSON);
  return output;
}



function sortDescending() {
  const sheet = SpreadsheetApp.openById('XXXX').getSheetByName('result');
  var range = sheet.getDataRange(); // データ範囲を取得
  range.sort({ column: 13, ascending: false }); // 1列目を降順にソート
}

function onEdit() {
  var sheet = SpreadsheetApp.openById('1vXnbUc81bfA3lWhDpGOS0YcRuUgxSn1HgcNYC7X-5Nw').getSheetByName('result');
  // チェックボックスONで起動
  if (sheet.getRange('M1').getValue()) {
    sortDescending();
  }
  sheet.getRange('M1').uncheck();
}
